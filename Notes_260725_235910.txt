    <script>
        let emiChartInstance = null;

            document.getElementById('targetDate').value = new Date().toISOString().split('T')[0];

            // Sync Inputs with Sliders
            setupRangeSync('emiAmount', 'emiAmountInput', calculateEmi);
            setupRangeSync('emiRate', 'emiRateInput', calculateEmi);
            setupRangeSync('emiTenure', 'emiTenureInput', calculateEmi);

            setupRangeSync('sipAmount', 'sipAmountInput', calculateSip);
            setupRangeSync('sipRate', 'sipRateInput', calculateSip);
            setupRangeSync('sipYears', 'sipYearsInput', calculateSip);

            // Trigger initial calculations
            calculateAge();
            calculateEmi();
            calculateSip();
            calculateGst();
            calculateBmi();
            calcPerc1();
            calcPerc2();
            calcPerc3();

            // Add input listeners for real-time recalculations
            document.getElementById('gstAmount').addEventListener('input', calculateGst);
            document.getElementById('bmiHeight').addEventListener('input', calculateBmi);
            document.getElementById('bmiWeight').addEventListener('input', calculateBmi);
        });

        // Tab Switcher with Animations
        function switchTab(tabId) {
            document.querySelectorAll('.calc-tab').forEach(el => {
                el.classList.add('hidden');
                el.classList.remove('animate-tab-content');
            });
            document.querySelectorAll('.tab-btn').forEach(el => {
                el.classList.remove('active', 'bg-blue-600', 'text-white');
                el.classList.add('text-slate-600');
            });

            const currentTab = document.getElementById(`tab-${tabId}`);
            currentTab.classList.remove('hidden');
            
            // Re-trigger animation
            void currentTab.offsetWidth;
            currentTab.classList.add('animate-tab-content');

            const activeBtn = document.getElementById(`btn-${tabId}`);
            activeBtn.classList.add('active');
            activeBtn.classList.remove('text-slate-600');
        }

        // Helper function to sync slider and text input
        function setupRangeSync(rangeId, inputId, calcFn) {
            const range = document.getElementById(rangeId);
            const input = document.getElementById(inputId);

            range.addEventListener('input', () => {
                input.value = range.value;
                calcFn();
            });

            input.addEventListener('change', () => {
                range.value = input.value;
                calcFn();
            });
        }

        // Currency Formatter
        function formatINR(val) {
            return new Intl.NumberFormat('en-IN', {
                style: 'currency',
                currency: 'INR',
                maximumFractionDigits: 0
            }).format(val);
        }

        // 1. AGE CALCULATOR LOGIC
        function calculateAge() {
            const dobVal = document.getElementById('dob').value;
            const targetVal = document.getElementById('targetDate').value;

            if (!dobVal) return;

            const dob = new Date(dobVal);
            const target = targetVal ? new Date(targetVal) : new Date();

            if (dob > target) {
                alert('Date of birth cannot be in the future!');
                return;
            }

            let years = target.getFullYear() - dob.getFullYear();
            let months = target.getMonth() - dob.getMonth();
            let days = target.getDate() - dob.getDate();

            if (days < 0) {
                months--;
                const lastMonth = new Date(target.getFullYear(), target.getMonth(), 0);
                days += lastMonth.getDate();
            }

            if (months < 0) {
                years--;
                months += 12;
            }

            document.getElementById('ageYears').textContent = years;
            document.getElementById('ageMonths').textContent = months;
            document.getElementById('ageDays').textContent = days;

            // Total aggregates
            const diffTime = Math.abs(target - dob);
            const totalDaysVal = Math.floor(diffTime / (1000 * 60 * 60 * 24));
            const totalMonthsVal = (years * 12) + months;
            const totalWeeksVal = Math.floor(totalDaysVal / 7);
            const totalHoursVal = totalDaysVal * 24;

            document.getElementById('totalMonths').textContent = totalMonthsVal.toLocaleString();
            document.getElementById('totalWeeks').textContent = totalWeeksVal.toLocaleString();
            document.getElementById('totalDays').textContent = totalDaysVal.toLocaleString();
            document.getElementById('totalHours').textContent = totalHoursVal.toLocaleString();

            // Next Birthday Calculation
            let nextBday = new Date(target.getFullYear(), dob.getMonth(), dob.getDate());
            if (target > nextBday) {
                nextBday.setFullYear(target.getFullYear() + 1);
            }
            const bdayDiffTime = nextBday - target;
            const bdayTotalDays = Math.ceil(bdayDiffTime / (1000 * 60 * 60 * 24));
            
            const bdayMonths = Math.floor(bdayTotalDays / 30.4375);
            const bdayDays = Math.floor(bdayTotalDays % 30.4375);

            document.getElementById('nextBirthday').textContent = `${bdayMonths} Months, ${bdayDays} Days`;
            document.getElementById('ageResult').classList.remove('hidden');
        }

        // 2. EMI CALCULATOR LOGIC
        function calculateEmi() {
            const P = parseFloat(document.getElementById('emiAmount').value);
            const annualRate = parseFloat(document.getElementById('emiRate').value);
            const r = (annualRate / 12) / 100;
            const n = parseFloat(document.getElementById('emiTenure').value) * 12;

            const emi = (P * r * Math.pow(1 + r, n)) / (Math.pow(1 + r, n) - 1);
            const totalPayment = emi * n;
            const totalInterest = totalPayment - P;

            document.getElementById('emiResult').textContent = formatINR(emi);
            document.getElementById('emiPrincipal').textContent = formatINR(P);
            document.getElementById('emiInterest').textContent = formatINR(totalInterest);
            document.getElementById('emiTotal').textContent = formatINR(totalPayment);

            renderEmiChart(P, totalInterest);
        }

        function renderEmiChart(principal, interest) {
            const ctx = document.getElementById('emiChart').getContext('2d');
            if (emiChartInstance) emiChartInstance.destroy();

            emiChartInstance = new Chart(ctx, {
                type: 'doughnut',
                data: {
                    labels: ['Principal', 'Interest'],
                    datasets: [{
                        data: [principal, interest],
                        backgroundColor: ['#2563eb', '#6366f1'],

                        hoverOffset: 6
                    }]

                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    animation: {
                        duration: 600,
                        easing: 'easeOutQuart'
                    },
                    plugins: {
                        legend: { position: 'bottom' }
                    }
                }
            });
        }

        // 3. SIP CALCULATOR LOGIC
        function calculateSip() {
            const P = parseFloat(document.getElementById('sipAmount').value);
            const i = (parseFloat(document.getElementById('sipRate').value) / 12) / 100;
            const n = parseFloat(document.getElementById('sipYears').value) * 12;

            const invested = P * n;
            const futureValue = P * ((Math.pow(1 + i, n) - 1) / i) * (1 + i);
            const returns = futureValue - invested;

            document.getElementById('sipTotalValue').textContent = formatINR(futureValue);
            document.getElementById('sipInvested').textContent = formatINR(invested);
            document.getElementById('sipReturns').textContent = formatINR(returns);

            renderSipChart(invested, returns);
        }

        function renderSipChart(invested, returns) {
            const ctx = document.getElementById('sipChart').getContext('2d');
            if (sipChartInstance) sipChartInstance.destroy();

            sipChartInstance = new Chart(ctx, {
                type: 'doughnut',
                data: {
                    labels: ['Invested', 'Est. Returns'],
                    datasets: [{
                        data: [invested, returns],
                        backgroundColor: ['#94a3b8', '#10b981'],
                        borderWidth: 0,
                        hoverOffset: 6
                    }]

                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    animation: {
                        duration: 600,
                        easing: 'easeOutQuart'

                        legend: { position: 'bottom' }
                    }

        }

        // 4. GST CALCULATOR LOGIC
        function setGstRate(rate) {
            selectedGstRate = rate;
            document.querySelectorAll('.gst-rate-btn').forEach(btn => {
                btn.classList.remove('active', 'bg-blue-600', 'text-white', 'shadow-md', 'shadow-blue-500/20');
                btn.classList.add('border', 'border-slate-200', 'hover:bg-slate-100', 'text-slate-700');
            });
            event.target.classList.add('active', 'bg-blue-600', 'text-white', 'shadow-md', 'shadow-blue-500/20');
            event.target.classList.remove('border', 'border-slate-200', 'hover:bg-slate-100', 'text-slate-700');
            calculateGst();
        }

        function calculateGst() {
            const amount = parseFloat(document.getElementById('gstAmount').value) || 0;
            const type = document.querySelector('input[name="gstType"]:checked').value;
            
            let gstTax = 0;
            let netAmount = 0;
            let totalAmount = 0;

            if (type === 'exclusive') {
                netAmount = amount;
                gstTax = (netAmount * selectedGstRate) / 100;
                totalAmount = netAmount + gstTax;
            } else {
                totalAmount = amount;
                netAmount = totalAmount * (100 / (100 + selectedGstRate));
                gstTax = totalAmount - netAmount;
            }

            const splitTax = gstTax / 2;

            document.getElementById('gstTotal').textContent = formatINR(totalAmount);
            document.getElementById('gstBase').textContent = formatINR(netAmount);
            document.getElementById('gstTax').textContent = formatINR(gstTax);
            document.getElementById('gstSplit').textContent = `${formatINR(splitTax)} / ${formatINR(splitTax)}`;
        }

        // 5. BMI CALCULATOR LOGIC
        function calculateBmi() {
            const heightCm = parseFloat(document.getElementById('bmiHeight').value);
            const weightKg = parseFloat(document.getElementById('bmiWeight').value);

            if (!heightCm || !weightKg || heightCm <= 0 || weightKg <= 0) return;

            const heightM = heightCm / 100;
            const bmi = weightKg / (heightM * heightM);

            const bmiScoreEl = document.getElementById('bmiScore');
            const bmiStatusEl = document.getElementById('bmiStatus');

            bmiScoreEl.textContent = bmi.toFixed(1);

            let status = 'Normal';
            let colorClass = 'bg-emerald-100 text-emerald-800';

            if (bmi < 18.5) {
                status = 'Underweight';
                colorClass = 'bg-blue-100 text-blue-800';
            } else if (bmi >= 18.5 && bmi <= 24.9) {
                status = 'Normal';
                colorClass = 'bg-emerald-100 text-emerald-800';
            } else if (bmi >= 25 && bmi <= 29.9) {
                status = 'Overweight';
                colorClass = 'bg-amber-100 text-amber-800';
            } else {
                status = 'Obesity';
                colorClass = 'bg-rose-100 text-rose-800';
            }

            bmiStatusEl.textContent = status;
            bmiStatusEl.className = `px-3 py-1 rounded-full text-xs font-bold ${colorClass} animate-pop`;
        }

        // 6. PERCENTAGE CALCULATOR LOGIC
        function calcPerc1() {
            const x = parseFloat(document.getElementById('perc1_x').value) || 0;
            const y = parseFloat(document.getElementById('perc1_y').value) || 0;
            const res = (x / 100) * y;
            document.getElementById('perc1_result').textContent = Number.isInteger(res) ? res : res.toFixed(2);
        }

        function calcPerc2() {
            const x = parseFloat(document.getElementById('perc2_x').value) || 0;
            const y = parseFloat(document.getElementById('perc2_y').value) || 0;
            if (y === 0) return;
            const res = (x / y) * 100;
            document.getElementById('perc2_result').textContent = `${res.toFixed(2)}%`;
        }

        function calcPerc3() {
            const x = parseFloat(document.getElementById('perc3_x').value) || 0;
            const y = parseFloat(document.getElementById('perc3_y').value) || 0;
            if (x === 0) return;
            const diff = y - x;
            const res = (diff / x) * 100;
            const sign = res >= 0 ? '+' : '';
            const el = document.getElementById('perc3_result');
            el.textContent = `${sign}${res.toFixed(2)}%`;
            el.className = `text-lg font-extrabold transition-all ${res >= 0 ? 'text-emerald-600' : 'text-rose-600'}`;
        }
